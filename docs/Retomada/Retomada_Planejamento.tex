\begin{frame}{O problema, com o que verificamos}
\begin{itemize}\setlength\itemsep{4pt}\small
  \item \alert{Cada retorno vira uma conversa nova.} A caixa do WhatsApp está com \code{lock_to_single_conversation=false} (Chatwoot 4.11.1). Na VM, 29 conversas têm exatamente 1 protocolo cada.
  \item \alert{O protocolo não guardava nada.} A tabela tinha só id, conversa, datas e quem encerrou; ao encerrar, processo, unidade e perfil eram apagados.
  \item \alert{A nota ao atendente era parcial.} Só saía quando a pessoa escolhia o canal; \emph{Falar com atendente} sem unidade transferia sem contexto.
  \item \alert{Balcão Virtual sem rastro.} A Júlia só mostrava o link: se a chamada caísse, a equipe não tinha telefone nem contexto.
  \item \alert{Resultado:} a pessoa repete o caso a cada atendente e a informação fica espalhada.
\end{itemize}
\end{frame}

\begin{frame}{O que não conseguimos confirmar}
\begin{itemize}\setlength\itemsep{5pt}\small
  \item \alert{Sem estudo} sobre retomada e resumo em tribunais brasileiros, nem estudo independente sobre resumo na transferência: a comparação é com norma de SAC e prática de mercado. \textbf{O piloto precisa medir.}
  \item O texto oficial da Res. CNJ 615/2025 não abriu direto (bloqueio do site); lemos uma transcrição. Conferir o art.~15 na versão compilada (Res.~674/2026).
  \item A Res.~615 não menciona chatbot nem atendimento: o enquadramento do resumo por IA e o anúncio ao usuário (art.~33) são \textbf{decisões a tomar com a área de IA do TJPA}.
  \item Números da Five9, Zendesk e Salesforce vieram por fontes secundárias; os relatórios originais não foram abertos.
  \item Chatwoot Captain (IA nativa) está nos planos pagos e enviaria as conversas a outro fornecedor: não é alternativa gratuita ao que construímos.
  \item A janela de 24 h do WhatsApp (retorno só com modelo aprovado) veio de documentação da Twilio: conferir na Meta.
\end{itemize}
\end{frame}

\begin{frame}{Retomada: como a Júlia decide}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{itemize}\setlength\itemsep{4pt}\scriptsize
  \item Menu: \textbf{\faHistory\ Retomar atendimento} antes de \emph{Falar com atendente}; o TKT digitado em qualquer mensagem também funciona.
  \item Dono = \code{sender.id} do contato do Chatwoot, gravado só de mensagem da própria pessoa.
  \item \textbf{Mesma resposta} para protocolo inexistente, de outro contato, sem dono (antigo ou apagado) e sem chat: não há como testar se um número existe.
  \item 3 tentativas e oferece o atendimento humano.
  \item \emph{Não sei o número}: lista só os protocolos \textbf{deste} contato, sem os restritos.
  \item Restaura perfil, unidade e processo \textbf{só onde estiverem vazios}; processo só se constar no acervo.
  \item TKT dentro de diligência ou relato continua sendo texto.
\end{itemize}
\end{column}
\begin{column}{0.44\textwidth}
\begin{tikzpicture}
  \node[draw=petroleo,rounded corners=4pt,fill=nevoa,inner sep=6pt,text width=0.95\textwidth,font=\scriptsize,align=left]{%
  \textbf{Limite conhecido}\\[2pt]
  Conferir o contato autentica o \emph{telefone}, não a \emph{pessoa}. Por isso a Júlia \textbf{não devolve conteúdo do caso} à pessoa: o histórico só vai para a equipe, que confirma a identidade antes de falar do caso.};
\end{tikzpicture}
\end{column}

\begin{frame}{Resumo para o atendente: fase 1 e fase 2}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Fase 1 (sem IA, sempre que há transferência)}
\begin{itemize}\setlength\itemsep{3pt}\scriptsize
  \item Nota privada, uma vez, \textbf{também em \emph{Falar com atendente}} sem unidade.
  \item Fatos do atendimento atual; protocolo vinculado; últimos 5 protocolos do contato, marcados \emph{vinculado pela pessoa} ou \emph{não confirmado}.
  \item Equipe, motivo, unidade e processo \emph{informados}, link da conversa anterior.
  \item \code{/encerrar <resumo>}: o atendente registra o que foi feito (até 600 caracteres).
  \item Erro no histórico não impede a transferência.
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Fase 2 (IA, atrás de chave)}
\begin{itemize}\setlength\itemsep{3pt}\scriptsize
  \item Depois da transferência, em segundo plano, \textbf{uma tentativa por protocolo}.
  \item Só mensagens públicas, com CPF, telefone e e-mail mascarados; a conversa é tratada como \textbf{dado}, não como instrução.
  \item Saída em campos fechados; \textbf{número ou data fora do texto é descartado}.
  \item Nota rotulada \emph{rascunho gerado por IA}; só se a conversa ainda está com o atendente.
  \item \alert{Nunca} para caso restrito, sem chat ou da Ouvidoria. Falha da IA não afeta nada.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Balcão Virtual: telefone de retorno e protocolo}
\begin{itemize}\setlength\itemsep{5pt}\small
  \item Antes do link, \emph{uma vez por protocolo}: \textbf{Posso retornar por este WhatsApp se a chamada cair?} Opções: neste número, outro número (validado com DDD) ou prefiro não informar.
  \item A Júlia grava uma \textbf{nota privada} com o protocolo e o telefone, \textbf{sem transferir}, e pede que a pessoa informe o TKT ao atendente do Balcão.
  \item Se a chamada cair, a pessoa volta e toca em \emph{Retomar atendimento}.
  \item O telefone não vai para log nem auditoria e é apagado com os fatos, em 90 dias.
  \item \alert{A confirmar:} quem atende o Balcão olha o Chatwoot? Se não olha, a nota só ajuda quem retorna por ele.
  \item \alert{A confirmar:} retorno depois de 24 h da última mensagem da pessoa exige modelo de mensagem aprovado na Meta.
\end{itemize}
\end{frame}

\begin{frame}{Riscos, limites e o que vamos medir}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\textbf{Riscos}
\begin{itemize}\setlength\itemsep{3pt}\scriptsize
  \item \alert{Retenção:} 90 dias de dados ligados ao contato; D7 segue com a DPO.
  \item \alert{Celular compartilhado:} o contato autentica o telefone, não a pessoa.
  \item \alert{Resumo por IA:} envia conversas a fornecedor externo; enquadrar na Res.~615 e decidir o anúncio ao usuário.
  \item \alert{Adoção:} sem \code{/encerrar <resumo>}, o valor está nos links.
  \item Resumo por IA lê só a última página da API (20 mensagens).
\end{itemize}
\end{column}
\begin{column}{0.44\textwidth}
\textbf{Métricas do piloto}
\begin{itemize}\setlength\itemsep{3pt}\scriptsize
  \item Transferências com nota.
  \item Retomadas concluídas e não localizadas.
  \item Por amostra com atendentes: quantas vezes a pessoa precisou \textbf{repetir} o caso e se a nota bastou.
  \item Uso e correções do rascunho de IA.
  \item Não há linha de base: medir antes de ligar.
\end{itemize}
\end{column}
\end{columns}
\end{frame}
